// Planet.cpp - 新建
FIntPoint APlanet::WorldToLatLonCoordinates(const FVector& WorldPosition) const
{
    // 球面坐标反算
    float Lat = FMath::Asin(WorldPosition.Y / Radius);
    float Lon = FMath::Atan2(WorldPosition.Z, WorldPosition.X);
    
    int32 LatIdx = FMath::FloorToInt((Lat - FMath::DegreesToRadians(MinLat)) / FMath::DegreesToRadians(LatStep));
    int32 LonIdx = FMath::FloorToInt(Lon / FMath::DegreesToRadians(LonStep));
    
    return FIntPoint(LatIdx, LonIdx);
}

FVector APlanet::LatLonToWorldCoordinates(int32 LatIdx, int32 LonIdx) const
{
    float LatRad = FMath::DegreesToRadians(MinLat) + LatIdx * FMath::DegreesToRadians(LatStep);
    float LonRad = LonIdx * FMath::DegreesToRadians(LonStep);
    
    return FVector(
        Radius * FMath::Cos(LatRad) * FMath::Cos(LonRad),
        Radius * FMath::Sin(LatRad),
        Radius * FMath::Cos(LatRad) * FMath::Sin(LonRad)
    );
}

// 实例索引 ↔ 经纬索引转换
FIntPoint APlanet::InstanceIndexToLatLon(int32 InstanceIndex) const
{
    int32 LatIdx = InstanceIndex / NumLon;
    int32 LonIdx = InstanceIndex % NumLon;
    return FIntPoint(LatIdx, LonIdx);
}

int32 APlanet::LatLonToInstanceIndex(int32 LatIdx, int32 LonIdx) const
{
    return LatIdx * NumLon + LonIdx;
}
```

    FHitResult Hit;

// Planet.cpp - 新建
void APlanet::GeneratePlots()
{
    if (!PlotHISM || !PlotHISM->GetStaticMesh()) return;

    PlotHISM->ClearInstances();

    float LatRadMin = FMath::DegreesToRadians(MinLat);
    float LatRadMax = FMath::DegreesToRadians(MaxLat);
    float LatRadStep = FMath::DegreesToRadians(LatStep);
    float LonRadStep = FMath::DegreesToRadians(LonStep);

    for (float LatRad = LatRadMin; LatRad <= LatRadMax; LatRad += LatRadStep)
    {
        for (float LonRad = 0; LonRad < 2*PI; LonRad += LonRadStep)
        {
            // 球面坐标
            FVector SurfacePoint = LatLonToWorldCoordinates(LatRad, LonRad);
            FVector Normal = SurfacePoint.GetSafeNormal();
            FQuat Rotation = FQuat::FindBetweenNormals(FVector::UpVector, Normal);
            
            FTransform Transform(Rotation, SurfacePoint, FVector::OneVector);
            PlotHISM->AddInstance(Transform);
        }
    }
}
```

// Planet.cpp - 新建
void APlanet::LoadFromSaveData(const TArray<FFarmLandPlotSaveGameData>& SaveData)
{
    // 1. 重新生成 HISM 实例（位置由算法确定）
    GeneratePlots();
    
    // 2. 恢复地块状态
    for (const auto& PlotSave : SaveData)
    {
        int32 PlotIndex = LatLonToInstanceIndex(PlotSave.LatIndex, PlotSave.LonIndex);
        PlotData[PlotIndex].StateData = PlotSave.StateData;
        PlotData[PlotIndex].MoisturePercentage = PlotSave.MoisturePercentage;
        PlotData[PlotIndex].FertilityPercentage = PlotSave.FertilityPercentage;
        
        // 3. 更新可视化
        UpdatePlotVisual(FIntPoint(PlotSave.LatIndex, PlotSave.LonIndex));
    }
}
